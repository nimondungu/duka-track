@app.route("/")
def index():
    conn = get_db()
    cursor = conn.cursor()

    cursor.execute("SELECT * FROM view_current_stock ORDER BY name ASC;")
    items = cursor.fetchall()

    # Sum today's cash vs mpesa
    cursor.execute("""
        SELECT 
            COALESCE(SUM(CASE WHEN payment_method = 'CASH' THEN total_amount ELSE 0 END), 0) as cash_total,
            COALESCE(SUM(CASE WHEN payment_method = 'MPESA' THEN total_amount ELSE 0 END), 0) as mpesa_total
        FROM transactions
        WHERE movement_type = 'OUT' AND DATE(timestamp, 'localtime') = DATE('now', 'localtime');
    """)
    totals = cursor.fetchone()

    conn.close()
    return render_template_string(
        HTML_TEMPLATE,
        items=items,
        today_cash=totals["cash_total"],
        today_mpesa=totals["mpesa_total"]
    )


@app.route("/sale", methods=["POST"])
def record_sale():
    item_id = int(request.form["item_id"])
    qty = int(request.form["quantity"])
    payment = request.form["payment_method"]

    conn = get_db()
    cursor = conn.cursor()

    # Verify live stock balance
    cursor.execute("SELECT name, unit_price, current_stock FROM view_current_stock WHERE item_id = ?", (item_id,))
    item = cursor.fetchone()

    if not item or item["current_stock"] < qty:
        flash(f"Insufficient stock for {item['name']}! Available: {item['current_stock'] if item else 0}", "error")
        conn.close()
        return redirect(url_for("index"))

    total = qty * item["unit_price"]
    cursor.execute("""
        INSERT INTO transactions (item_id, movement_type, payment_method, quantity, unit_price, total_amount)
        VALUES (?, 'OUT', ?, ?, ?, ?)
    """, (item_id, payment, qty, item["unit_price"], total))

    conn.commit()
    conn.close()
    flash(f"Recorded sale: {qty}x {item['name']} via {payment} (KES {total:,.1f})", "success")
    return redirect(url_for("index"))


@app.route("/restock", methods=["POST"])
def restock():
    item_id = int(request.form["item_id"])
    qty = int(request.form["quantity"])

    conn = get_db()
    cursor = conn.cursor()
    cursor.execute("SELECT name, unit_price FROM items WHERE item_id = ?", (item_id,))
    item = cursor.fetchone()

    if item:
        total = qty * item["unit_price"]
        cursor.execute("""
            INSERT INTO transactions (item_id, movement_type, payment_method, quantity, unit_price, total_amount)
            VALUES (?, 'IN', 'N/A', ?, ?, ?)
        """, (item_id, qty, item["unit_price"], total))
        conn.commit()
        flash(f"Added restock: +{qty} units for {item['name']}", "success")

    conn.close()
    return redirect(url_for("index"))
